# Sem este arquivo o site é rastreado assim mesmo, mas o buscador
# perde tempo entrando no app (que exige login e não tem nada a
# indexar) e não sabe onde fica o mapa do site.

User-agent: *
Allow: /

# O app é área logada: toda URL abaixo de /app/ responde a mesma
# casca vazia para quem não tem sessão. Indexar isso colocaria no
# Google uma página em branco com o nome do produto — pior do que
# não ter página nenhuma.
Disallow: /app/

Sitemap: https://workap.com.br/sitemap.xml
